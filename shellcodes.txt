Stop IIS and free port 80:


Stop-Service -Name W3SVC
Set-Service -Name W3SVC -StartupType Disabled
Then restart Docker containers:


docker compose down
docker compose up -d


for docker to pull repo from github:

git pull origin main
docker compose down
docker compose up -d --build

cd C:\Users\Administrator\Desktop\scholarship
git pull origin main
docker compose build --no-cache server
docker compose up -d --force-recreate server

docker compose build --no-cache client


firewall rule: 
netsh advfirewall firewall add rule name="Scholarship Frontend" dir=in action=allow protocol=TCP localport=80
